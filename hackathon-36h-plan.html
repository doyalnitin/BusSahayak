3. VOICE ENGINE - Build This First
3.1 Architecture (3-Layer Stack)
Layer 1: STT (Speech to Text)
Primary: Web Speech API (browser built-in)
Fallback: Vosk (offline, self-hosted)
Format: 16kHz PCM audio
Latency: 50-200ms
Languages: Hindi (hi-IN), English (en-US)
Code:
webkitSpeechRecognition
SpeechRecognition.lang = "hi-IN"
SpeechRecognition.continuous = false
Layer 2: LLM (Understanding)
Primary: Regex intent classifier (instant)
Fallback: Gemma 2B Q4 (local, 150ms)
Pattern matching:
/Mumbai.*Delhi/ = city pair
/kal|parson/ = relative date
/AC|sleeper/ = bus preference
/window|aisle/ = seat preference
Latency: 5-150ms
Layer 3: TTS (Text to Speech)
Primary: Web Speech Synthesis (browser)
Fallback: Piper (local, 50ms)
Format: 24kHz audio output
Latency: 50-200ms
Voice: Hindi female voice
Code:
speechSynthesis.speak(utterance)
utterance.lang = "hi-IN"
3.2 Voice Hooks - Complete Code
// hooks/useVoiceInput.ts - Speech Recognition import { useState, useCallback, useRef } from 'react'; export function useVoiceInput() { const [isListening, setIsListening] = useState(false); const [transcript, setTranscript] = useState(''); const recognitionRef = useRef(null); const startListening = useCallback(() => { const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SpeechRecognition) { alert("Voice not supported"); return; } const recognition = new SpeechRecognition(); recognition.lang = 'hi-IN'; // Hindi primary recognition.continuous = false; recognition.interimResults = true; recognition.maxAlternatives = 1; recognition.onstart = () => setIsListening(true); recognition.onresult = (event) => { const result = event.results[0][0].transcript; setTranscript(result); }; recognition.onend = () => setIsListening(false); recognition.onerror = (e) => { console.error(e); setIsListening(false); }; recognitionRef.current = recognition; recognition.start(); }, []); const stopListening = useCallback(() => { recognitionRef.current?.stop(); }, []); return { isListening, transcript, startListening, stopListening }; } // hooks/useVoiceOutput.ts - Speech Synthesis export function useVoiceOutput() { const speak = useCallback((text: string, lang = 'hi-IN') => { const utterance = new SpeechSynthesisUtterance(text); utterance.lang = lang; utterance.rate = 1.0; utterance.pitch = 1.0; utterance.volume = 1.0; window.speechSynthesis.speak(utterance); }, []); return { speak }; }
3.3 Intent Classifier - Pattern Matching
// services/intentClassifier.ts const CITIES = ['Mumbai','Delhi','Bangalore','Chennai','Kolkata', 'Pune','Hyderabad','Ahmedabad','Jaipur','Lucknow', 'Guwahati','Shillong','Imphal','Agartala']; const CITY_HINDI = { 'Mumbai': ['मुंबई','बॉम्बे'], 'Delhi': ['दिल्ली','नई दिल्ली'], 'Bangalore': ['बैंगलोर','बेंगलुरु'], 'Chennai': ['चेन्नई'], 'Guwahati': ['गुवाहाटी'], 'Shillong': ['शिलोंग'] }; export function classifyIntent(text: string) { const t = text.toLowerCase(); let from = null, to = null, date = null, busType = null; // Extract cities (English + Hindi) for (const city of CITIES) { if (t.includes(city.toLowerCase())) { if (!from) from = city; else if (!to) to = city; } for (const h of (CITY_HINDI[city] || [])) { if (t.includes(h)) { if (!from) from = city; else if (!to) to = city; } } } // Extract patterns: "X se Y" / "X to Y" / "X Y jana hai" const seMatch = t.match(/(\w+)\s+se\s+(\w+)/); if (seMatch) { from = seMatch[1]; to = seMatch[2]; } // Extract date if (t.includes('kal')) date = 'tomorrow'; else if (t.includes('parson')) date = 'day_after'; else if (t.includes('aaj')) date = 'today'; // Extract bus type if (t.includes('ac') || t.includes('एसी')) busType = 'AC'; else if (t.includes('sleeper')) busType = 'Sleeper'; else if (t.includes('non ac') || t.includes('non-ac')) busType = 'Non-AC'; return { from, to, date, busType, raw: text }; }

6. UX/UI DESIGN - Best Practices for Blind Users
6.1 Design Principles
Touch Target Size
Minimum 48x48dp touch targets
Large buttons (64x64dp minimum)
Generous spacing (16dp between elements)
No tiny links or close buttons
Rule: If finger can't hit it, blind user definitely can't
Color Contrast
WCAG AAA compliance (7:1 ratio)
White background (#ffffff)
Primary blue (#2563eb)
Dark text (#1e293b)
Rule: Low vision users need high contrast
Voice-First Flow
Every action has voice equivalent
Confirmation before destructive actions
Read screen content automatically
Number-based selection (say "1" to pick)
Rule: Never require sight to complete a task
