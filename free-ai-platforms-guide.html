// ScreenReaderService.kt — Capture screen elements
class ScreenReaderService : AccessibilityService() {

    override fun onAccessibilityEvent(event: AccessibilityEvent) {
        val rootNode = rootInActiveWindow ?: return

        // Extract all elements from screen
        val elements = mutableListOf<ScreenElement>()
        traverseNode(rootNode, elements)

        // Send to Python backend for processing
        sendToBackend(elements)
    }

    private fun traverseNode(node: AccessibilityNodeInfo, elements: MutableList<ScreenElement>) {
        val bounds = Rect()
        node.getBoundsInScreen(bounds)

        val element = ScreenElement(
            text = node.text?.toString() ?: "",
            className = node.className?.toString() ?: "",
            isClickable = node.isClickable,
            isScrollable = node.isScrollable,
            bounds = Bounds(bounds.left, bounds.top, bounds.right, bounds.bottom),
            role = detectRole(node)
        )
        elements.add(element)

        // Recurse into children
        for (i in 0 until node.childCount) {
            node.getChild(i)?.let { traverseNode(it, elements) }
        }
    }

    private fun detectRole(node: AccessibilityNodeInfo): String {
        return when {
            node.className?.toString()?.contains("Button") == true -> "button"
            node.className?.toString()?.contains("EditText") == true -> "input"
            node.className?.toString()?.contains("TextView") == true -> "text"
            node.isClickable -> "clickable"
            else -> "other"
        }
    }

    // Execute actions on screen elements
    fun performClick(node: AccessibilityNodeInfo) {
        node.performAction(AccessibilityNodeInfo.ACTION_CLICK)
    }

    fun performScroll(node: AccessibilityNodeInfo) {
        node.performAction(AccessibilityNodeInfo.ACTION_SCROLL_FORWARD)
    }

    fun performType(node: AccessibilityNodeInfo, text: String) {
        val args = Bundle().apply {
            putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, text)
        }
        node.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args)
    }
}

import base64
from PIL import Image
from io import BytesIO
import subprocess

def capture_screen() -> str:
    """Capture Android screen and return base64 encoded image"""
    # Method 1: Using adb screencap (requires ADB connection)
    result = subprocess.run(
        ['adb', 'exec-out', 'screencap', '-p'],
        capture_output=True
    )

    # Convert PNG bytes to base64
    image_bytes = result.stdout
    base64_encoded = base64.b64encode(image_bytes).decode('utf-8')
    return base64_encoded

def capture_screen_from_accessibility(elements: list) -> str:
    """Create a virtual screen representation from accessibility nodes"""
    # Build a simplified representation of the screen
    screen_data = {
        "elements": [
            {
                "text": el.text,
                "role": el.role,
                "bounds": el.bounds,
                "clickable": el.is_clickable
            }
            for el in elements
        ]
    }
    return json.dumps(screen_data)

def extract_text_from_elements(elements: list) -> str:
    """Extract all text from accessibility elements"""
    texts = [el.text for el in elements if el.text.strip()]
    return "\n".join(texts)

import google.generativeai as genai
import base64, time

genai.configure(api_key="YOUR_GEMINI_API_KEY")
model = genai.GenerativeModel("gemini-2.0-flash")

def analyze_screen(screenshot_base64: str, ocr_text: str) -> dict:
    """Use Gemini Vision to understand the screen"""
    start = time.time()

    prompt = f"""Analyze this mobile screen screenshot.
    OCR extracted text:
    {ocr_text}

    Please provide:
    1. Screen summary in Hindi (2-3 lines)
    2. List of all buttons/links with their text and position
    3. List of input fields
    4. Suggested actions user can take
    5. Which element should be focused first?

    Return as JSON with keys: summary, buttons, inputs, actions, focus_first
    """

    # Send image + text to Gemini Vision
    image_data = base64.b64decode(screenshot_base64)
    response = model.generate_content([
        prompt,
        {"mime_type": "image/png", "data": image_data}
    ])

    latency = (time.time() - start) * 1000
    print(f"Gemini Vision latency: {latency:.0f}ms")

    return {
        "response": response.text,
        "latency_ms": latency
    }

def find_clickable_elements(elements: list) -> list:
    """Find buttons and clickable elements"""
    clickable = []
    for el in elements:
        if el.is_clickable or el.role in ["button", "link"]:
            clickable.append({
                "text": el.text,
                "role": el.role,
                "bounds": el.bounds,
                "center": {
                    "x": (el.bounds.left + el.bounds.right) // 2,
                    "y": (el.bounds.top + el.bounds.bottom) // 2
                }
            })
    return clickable

Action Executor — Click, Type, Scroll (Python + ADB)
import subprocess, json

class ActionExecutor:
    def __init__(self):
        self.commands = []

    def tap(self, x: int, y: int):
        """Tap at coordinates"""
        subprocess.run(['adb', 'shell', 'input', 'tap', str(x), str(y)])
        print(f"Tapped at ({x}, {y})")

    def type_text(self, text: str):
        """Type text via ADB"""
        subprocess.run(['adb', 'shell', 'input', 'text', text])

    def scroll_down(self):
        """Scroll down"""
        subprocess.run(['adb', 'shell', 'input', 'swipe',
                        '540', '1500', '540', '500', '300'])

    def scroll_up(self):
        """Scroll up"""
        subprocess.run(['adb', 'shell', 'input', 'swipe',
                        '540', '500', '540', '1500', '300'])

    def go_back(self):
        """Press back button"""
        subprocess.run(['adb', 'shell', 'input', 'keyevent', '4'])

    def find_and_tap(self, elements: list, target_text: str):
        """Find element by text and tap it"""
        for el in elements:
            if target_text.lower() in el.text.lower():
                center = el.center
                self.tap(center['x'], center['y'])
                return True
        return False

    def execute_command(self, command: str, elements: list):
        """Execute voice command on screen"""
        cmd = command.lower()
        if "click" in cmd or "tap" in cmd:
            # Extract target from command
            target = cmd.replace("click", "").replace("tap", "").strip()
            self.find_and_tap(elements, target)
        elif "scroll down" in cmd:
            self.scroll_down()
        elif "scroll up" in cmd:
            self.scroll_up()
        elif "back" in cmd:
            self.go_back()
        elif "type" in cmd:
            text = cmd.replace("type", "").strip()
            self.type_text(text)

from groq import Groq
import time

client = Groq(api_key="YOUR_GROQ_API_KEY")

def transcribe_audio(audio_file_path: str) -> dict:
    """Transcribe audio using Groq Whisper — ~80ms"""
    start = time.time()

    with open(audio_file_path, "rb") as audio_file:
        transcription = client.audio.transcriptions.create(
            file=("audio.wav", audio_file.read()),
            model="whisper-large-v3",
            language="hi",  # Hindi
            response_format="verbose_json"
        )

    latency = (time.time() - start) * 1000
    print(f"STT latency: {latency:.0f}ms")
    print(f"Text: {transcription.text}")

    return {
        "text": transcription.text,
        "latency_ms": latency,
        "language": transcription.language
    }

from groq import Groq
import time, json

client = Groq(api_key="YOUR_GROQ_API_KEY")

SYSTEM_PROMPT = """You are BusSahayak, a Hindi voice assistant for bus booking.
Always respond in Hindi. Be concise (1-2 sentences).
Available actions: search_buses, book_seat, cancel_booking, show_status.
Return JSON: {"action": "...", "params": {...}, "reply": "..."}"""

def process_command(text: str) -> dict:
    """Process voice command using Groq Gemma — ~100ms"""
    start = time.time()

    response = client.chat.completions.create(
        model="gemma2-9b-it",
        messages=[
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": text}
        ],
        temperature=0.3,
        max_tokens=150
    )

    latency = (time.time() - start) * 1000
    reply = response.choices[0].message.content

    print(f"LLM latency: {latency:.0f}ms")

    try:
        parsed = json.loads(reply)
    except:
        parsed = {"action": "unknown", "reply": reply}

    return {
        "action": parsed.get("action", "none"),
        "params": parsed.get("params", {}),
        "reply": parsed.get("reply", reply),
        "latency_ms": latency
    }

# Usage
result = process_command("मुझे दिल्ली से मुंबई की बस ढूंढो")
