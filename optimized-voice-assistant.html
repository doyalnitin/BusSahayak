5. Complete Optimized Pipeline — Production-Ready Python Script
voice_assistant_optimized.py
Full Pipeline: VOSK → Gemma (llama.cpp) → Piper → Playback python 
import time, json, queue, subprocess, tempfile, os, sys
import sounddevice as sd
from vosk import Model, KaldiRecognizer

# ============== CONFIG ==============
VOSK_MODEL = "./vosk-model-small-hi-0.22"
LLAMA_CPP  = "./llama.cpp/main"
GEMMA_MODEL = "./gemma-2b.Q4_0.gguf"
PIPER_MODEL = "./hi_IN-swara-small.onnx"
SAMPLE_RATE = 16000
BLOCK_SIZE  = 4000  # 250ms chunks — faster response

# ============== LATENCY TRACKER ==============
class LatencyTracker:
    def __init__(self):
        self.times = {}
        self.results = []

    def start(self, name):
        self.times[name] = time.time()

    def end(self, name):
        if name in self.times:
            ms = (time.time() - self.times[name]) * 1000
            self.results.append((name, ms))
            print(f"  ⏱ {name}: {ms:.0f}ms")
            return ms
        return 0

    def total(self):
        return sum(ms for _, ms in self.results)

    def summary(self):
        total_ms = self.total()
        print(f"\n{'='*50}")
        print(f"  TOTAL LATENCY: {total_ms:.0f}ms")
        print(f"  Status: {'✓ PASS' if total_ms < 500 else '✗ FAIL'} (<500ms target)")
        print(f"{'='*50}\n")
        self.results.clear()

tracker = LatencyTracker()

# ============== VOSK (OPTIMIZED) ==============
print("Loading VOSK model...")
model = Model(VOSK_MODEL)
rec = KaldiRecognizer(model, SAMPLE_RATE)
rec.SetWords(True)
q = queue.Queue()

def callback(indata, frames, time_info, status):
    q.put(bytes(indata))

def listen():
    tracker.start("stt")
    with sd.RawInputStream(
        samplerate=SAMPLE_RATE,
        blocksize=BLOCK_SIZE,
        dtype='int16',
        channels=1,
        callback=callback
    ):
        while True:
            data = q.get()
            if rec.AcceptWaveform(data):
                result = json.loads(rec.Result())
                text = result.get('text', '').strip()
                tracker.end("stt")
                return text

# ============== GEMMA (QUANTIZED via llama.cpp) ==============
def ask_gemma(prompt):
    tracker.start("llm")
    cmd = [
        LLAMA_CPP,
        "-m", GEMMA_MODEL,
        "-n", "64",         # max tokens (keep short for speed)
        "--temp", "0.7",
        "--threads", "4",   # use all CPU cores
        "-p", f"""You are a Hindi bus assistant. Reply in 1-2 sentences.

User: {prompt}
Assistant:"""
    ]
    try:
        result = subprocess.run(
            cmd, capture_output=True,
            text=True, timeout=10
        )
        response = result.stdout.strip()
        # Clean up: remove prompt echo and artifacts
        lines = response.split('\n')
        for line in lines:
            if line.strip() and not line.startswith(' '):
                response = line.strip()
                break
    except subprocess.TimeoutExpired:
        response = "Maaf kijiye, jawab dene mein samay lag raha hai."
    tracker.end("llm")
    return response

# ============== PIPER (OPTIMIZED with streaming) ==============
def speak(text):
    tracker.start("tts")
    try:
        # Stream audio directly — no temp file
        process = subprocess.Popen(
            ["piper", "--model", PIPER_MODEL, "--output-raw"],
            stdin=subprocess.PIPE, stdout=subprocess.PIPE
        )
        audio_data, _ = process.communicate(input=text.encode("utf-8"))
        tracker.end("tts")

        # Play audio
        tracker.start("playback")
        player = subprocess.Popen(
            ["afplay", "-r", "22050", "-f", "S16_LE", "-c", "1", "-"],
            stdin=subprocess.PIPE
        )
        player.communicate(input=audio_data)
        tracker.end("playback")
    except Exception as e:
        print(f"TTS error: {e}")
        tracker.end("tts")

# ============== CONVERSATION HISTORY ==============
conversation_history = []

def build_prompt(user_text):
    history = "\n".join(conversation_history[-6:])
    return f"""You are a Hindi bus assistant. Reply in 1-2 sentences.

{history}
User: {user_text}
Assistant:"""

# ============== MAIN ==============
def main():
    print("🎤 Voice Assistant Ready (<500ms target)")
    print("Say 'bye' to exit\n")

    while True:
        print("Listening...", end=" ", flush=True)
        text = listen()

        if not text:
            continue
        print(f"You said: {text}")

        if "bye" in text.lower():
            speak("Alvida! Safe journey!")
            break

        # Build contextual prompt
        prompt = build_prompt(text)
        response = ask_gemma(prompt)

        # Store in history
        conversation_history.append(f"User: {text}")
        conversation_history.append(f"Assistant: {response}")

        print(f"Assistant: {response}")
        speak(response)

        tracker.summary()

if __name__ == "__main__":
    main()

8. Conversation Threading — Maintaining Context at Speed
Context Window Strategy

Challenge: Keeping conversation context while maintaining <500ms latency.


Solution: Keep only the last N exchanges in the prompt. More history = slower inference.


Optimized Conversation Manager python 
# Keep last N exchanges for context
conversation_history = []
MAX_HISTORY = 3  # 3 exchanges = 6 lines max

def build_prompt(user_text):
    # Only last 3 exchanges = ~50 tokens overhead
    history = "\n".join(conversation_history[-(MAX_HISTORY * 2):])
    return f"""You are a Hindi bus assistant.
Reply in 1-2 sentences. Be concise.

{history}
User: {user_text}
Assistant:"""

def update_history(user_text, response):
    conversation_history.append(f"User: {user_text}")
    conversation_history.append(f"Assistant: {response}")
    # Keep only last N exchanges
    if len(conversation_history) > MAX_HISTORY * 2:
        conversation_history.pop(0)
        conversation_history.pop(0)
Context vs Latency Tradeoff
History Size Extra Tokens Extra Latency Context Quality
1 exchange ★ ~20 tokens +20ms Basic
3 exchanges ★ ~50 tokens +50ms Good balance
5 exchanges ~100 tokens +100ms Good
10 exchanges ~200 tokens +200ms Excellent
Recommendation: Use 3 exchanges (6 turns) for the best balance. This adds only ~50ms to LLM latency while providing enough context for coherent conversations.

Model Not Loading
Issue: "model not found" error
Fix: Check file path is correct. Use absolute paths.
Issue: "out of memory"
Fix: Use Q4_0 (1.5GB) instead of FP16 (4GB). Close other apps.
Issue: "shared library not found"
Fix: Rebuild llama.cpp: make clean && make LLAMA_METAL=1
Issue: VOSK model download fails
Fix: Download manually from alphacephei.com/vosk/models
Slow Inference Fixes
LLM too slow: Use Q4_0 quantization, limit tokens to 64, reduce threads
STT too slow: Use small VOSK model (50MB), reduce blocksize to 4000
TTS too slow: Use small Piper model, stream audio with --output-raw
General: Close background apps, ensure CPU isn't throttled, use GPU if available
