3. Full Backend Code — Complete Flow
main.py — FastAPI Complete Backend 
# main.py — BusSahayak Backend
from fastapi import FastAPI, WebSocket
from google import genai
from google.genai import types
import psycopg2
import redis
import httpx
import json
from datetime import datetime, timedelta

app = FastAPI()
client = genai.Client(api_key="YOUR_GEMINI_KEY")
r = redis.Redis(host='localhost', port=6379, decode_responses=True)
db = psycopg2.connect(host="localhost", database="bus_sahayak",
                       user="bussahayak", password="your_pass")

# Define Gemini tools
search_tool = types.FunctionDeclaration(
    name="search_buses",
    description="Search buses between two cities",
    parameters=types.Schema(
        type=types.Type.OBJECT,
        properties={
            "from_city": types.Schema(type=types.Type.STRING),
            "to_city": types.Schema(type=types.Type.STRING),
            "date": types.Schema(type=types.Type.STRING),
        },
        required=["from_city", "to_city"]
    )
)

config = types.GenerateContentConfig(
    tools=[types.Tool(function_declarations=[search_tool])],
    system_instruction="""You are BusSahayak voice assistant.
    Speak in Hindi/Hinglish. Be friendly and helpful.
    When user asks for buses, call search_buses function.
    Read results in spoken format. Always ask 'Kya ye sahi hai?'"
)

@app.websocket("/voice")
async def voice_chat(websocket: WebSocket):
    await websocket.accept()
    while True:
        # Step 1: Receive audio from user
        audio = await websocket.receive_bytes()

        # Step 2-3: Send to Gemini → Get JSON
        response = client.models.generate_content(
            model="gemini-2.0-flash",
            contents=audio,
            config=config
        )

        # Step 4: Check if Gemini wants to call function
        if response.function_calls:
            args = response.function_calls[0].args

            # Step 5: Query PostgreSQL (cached seats)
            cur = db.cursor()
            cur.execute("""SELECT seat_number, seat_type, status, price
                FROM seat_cache WHERE bus_id IN
                (SELECT bus_id FROM zuelpay_cache
                WHERE from_city=%s AND to_city=%s AND date=%s)""",
                (args["from_city"], args["to_city"], args["date"]))
            cached_seats = cur.fetchall()

            # Step 6: Query ZuelPay API (live buses)
            async with httpx.AsyncClient() as client_http:
                zuelpay = await client_http.get(
                    "https://api.zuelpay.com/v1/buses",
                    params={"from": args["from_city"],
                            "to": args["to_city"],
                            "date": args["date"]}
                )
                live_buses = zuelpay.json()

            # Step 7: Merge PostgreSQL + ZuelPay
            merged = merge_data(live_buses, cached_seats)

            # Step 8: Send merged data to Gemini
            final_prompt = f"""User wants: {args}
Available buses: {json.dumps(merged)}
Generate spoken Hindi response. Ask 'Kya ye sahi hai?'"

            voice_response = client.models.generate_content(
                model="gemini-2.0-flash",
                contents=final_prompt
            )

            # Step 9: Send voice back to user
            await websocket.send_text(voice_response.text)

        else:
            # Normal chat — no function call needed
            await websocket.send_text(response.text)

def merge_data(live_buses, cached_seats):
    """Merge ZuelPay live data with PostgreSQL cache"""
    merged = []
    for bus in live_buses:
        bus_seats = [s for s in cached_seats if s[0] == bus["bus_id"]]
        bus["available_seats"] = len([s for s in bus_seats if s[2] == "available"])
        bus["window_available"] = any(s[1] == "window" and s[2] == "available" for s in bus_seats)
        merged.append(bus)
    return merged
4. Data at Each Step — What Changes
Step Component Input Output Data Format
1 Microphone User voice 16kHz PCM audio bytes bytes
2 WebSocket Audio bytes Forwarded to Gemini WebSocket frame
3 Gemini LLM Audio stream Structured JSON (function call) FunctionCall
4 Backend FunctionCall JSON Parsed args (city, date, seat) dict
5 PostgreSQL SQL query (bus_id) Cached seat matrix tuple[]
6 ZuelPay API HTTP params (from, to, date) Live bus list with prices JSON
7 Backend Merge PostgreSQL + ZuelPay data Merged bus list with seat status JSON
8 Gemini LLM Merged data + prompt Natural Hindi response text
9 Speaker Hindi text "Kya ye sahi hai?" audio

Key Insight — Ye Flow Kya Karta Hai:

Gemini = Translator
Voice → JSON (understands user) + JSON → Voice (speaks back). Two jobs, one API.
Backend = Merger
Combines PostgreSQL (cached seats) + ZuelPay (live buses) into one merged dataset for Gemini.
PostgreSQL = Cache
Stores seat matrix from ZuelPay (refreshed every 5 min). Faster than hitting ZuelPay every time.
