import whisper
import ollama
from TTS.api import TTS

import numpy as np

# 1. Load YOUR trained models

# 2. Voice Pipeline
def listen_and_respond():
    audio = record_from_mic(duration=5)

    # STT: Your Custom Whisper (Hindi bus commands)
    text = stt_model.transcribe(audio, language="hi")["text"]
    print(f"You said: {text}")

    # LLM: Your Custom Gemma via Ollama
    response = ollama.chat(model="custom-gemma", messages=[{
        "role": "system",
        "content": "You are a Hindi bus booking assistant for blind users. Reply in Hindi."
    }, {
        "role": "user",
        "content": text
    }])
    reply = response["message"]["content"]
    print(f"Assistant: {reply}")

    # TTS: Your Custom Coqui (YOUR voice)
    tts_model.tts_to_file(text=reply, file_path="output.wav")
    play_audio("output.wav")

# 3. Screen Reader
def read_screen(screenshot_path):
    from llava.model.builder import load_pretrained_model
    model = load_pretrained_model("llava-hf/llava-1.5-7b-hf")
    response = model.generate(
        image=screenshot_path,
        prompt="What's on this screen? List all buttons and actions in Hindi."
    )
    return response

whisper_finetune_colab.py
# STEP 1: Install Dependencies

# STEP 2: Mount Google Drive
from google.colab import drive
drive.mount('/content/drive')

# STEP 3: Load Dataset
import json, os, torch
from torch.utils.data import Dataset, DataLoader
import whisper, torchaudio

class HindiBusDataset(Dataset):
    def __init__(self, jsonl_path, audio_dir):
        self.examples = []
        with open(jsonl_path, 'r', encoding='utf-8') as f:
            for line in f:
                data = json.loads(line.strip())
                audio_path = os.path.join(audio_dir, data["audio"])
                if os.path.exists(audio_path):
                    self.examples.append({"audio_path": audio_path, "text": data["text"]})

    def __len__(self): return len(self.examples)

    def __getitem__(self, idx):
        ex = self.examples[idx]
        audio, sr = torchaudio.load(ex["audio_path"])
        if sr != 16000:
            audio = torchaudio.transforms.Resample(sr, 16000)(audio)
        if audio.shape[0] > 1:
            audio = audio.mean(dim=0, keepdim=True)
        audio = audio.squeeze()
        max_len = 16000 * 30
        if audio.shape[0] > max_len:
            audio = audio[:max_len]
        else:
            audio = torch.nn.functional.pad(audio, (0, max_len - audio.shape[0]))
        return {"audio": audio, "text": ex["text"].lower().strip()}

dataset = HindiBusDataset("/content/drive/MyDrive/hindi_bus_commands/training_dataset.jsonl",
                           "/content/drive/MyDrive/hindi_bus_commands/")
print(f"Loaded {len(dataset)} examples")

# STEP 4: Load Whisper + Apply LoRA
from peft import LoraConfig, get_peft_model, TaskType

base_model = whisper.load_model("base")

lora_config = LoraConfig(
    r=8, lora_alpha=16,
    target_modules=["q_proj", "v_proj", "k_proj", "out_proj"],
    lora_dropout=0.1, bias="none",
    task_type=TaskType.CAUSAL_LM
)

model = get_peft_model(base_model, lora_config)
model.print_trainable_parameters()
# Output: trainable: ~2M | total: ~74M | trainable%: ~2.7%

# STEP 5: Training Loop
from torch.optim import AdamW
from torch.cuda.amp import autocast, GradScaler

optimizer = AdamW(model.parameters(), lr=1e-4)
scaler = GradScaler()
tokenizer = whisper.tokenizer.get_tokenizer(language="hi", multilingual=True)

def collate_fn(batch):
    return {
        "audio": torch.stack([b["audio"] for b in batch]),
        "texts": [b["text"] for b in batch]
    }

dataloader = DataLoader(dataset, batch_size=4, shuffle=True, collate_fn=collate_fn)
model.train()

for epoch in range(5):
    total_loss = 0
    for batch_idx, batch in enumerate(dataloader):
        optimizer.zero_grad()
        audio = batch["audio"].to("cuda")
        texts = batch["texts"]

        with autocast():
            audio_features = model.encoder(audio)
            text_tokens = [tokenizer.encode(t) for t in texts]
            loss = torch.tensor(0.0, requires_grad=True)
            for i, tokens in enumerate(text_tokens):
                target = torch.tensor(tokens).to("cuda")
                dec_out = model.decoder(
                    torch.tensor([[tokenizer.sot] + tokens[:-1]]).to("cuda"),
                    audio_features[i:i+1]
                )
                loss = loss + torch.nn.functional.cross_entropy(dec_out[:, -1, :], target)
            loss = loss / len(text_tokens)

        scaler.scale(loss).backward()
        scaler.step(optimizer)
        scaler.update()
        total_loss += loss.item()

        if batch_idx % 10 == 0:
            print(f"Epoch {epoch+1}, Batch {batch_idx}, Loss: {loss.item():.4f}")

    print(f"Epoch {epoch+1} Complete. Avg Loss: {total_loss/len(dataloader):.4f}")

# STEP 6: Save Model
save_path = "/content/drive/MyDrive/custom-whisper-hindi/"
os.makedirs(save_path, exist_ok=True)
model.save_pretrained(save_path)
torch.save(model.state_dict(), os.path.join(save_path, "whisper-hindi-bus.pt"))
print("Training complete!")

# STEP 1: Install Dependencies
!pip install -q transformers peft bitsandbytes accelerate datasets trl

# STEP 2: Load Gemma 2B with 4-bit Quantization
import torch
from transformers import AutoModelForCausalLM, AutoTokenizer, BitsAndBytesConfig, TrainingArguments
from peft import LoraConfig, get_peft_model, prepare_model_for_kbit_training
from trl import SFTTrainer
from datasets import load_dataset

bnb_config = BitsAndBytesConfig(
    load_in_4bit=True,
    bnb_4bit_quant_type="nf4",
    bnb_4bit_compute_dtype=torch.bfloat16,
    bnb_4bit_use_double_quant=True,
)

model = AutoModelForCausalLM.from_pretrained(
    "google/gemma-2b",
    quantization_config=bnb_config,
    device_map="auto",
    trust_remote_code=True
)
tokenizer = AutoTokenizer.from_pretrained("google/gemma-2b")
tokenizer.pad_token = tokenizer.eos_token
model = prepare_model_for_kbit_training(model)

# STEP 3: Configure LoRA
lora_config = LoraConfig(
    r=16, lora_alpha=32,
    target_modules=["q_proj", "v_proj", "k_proj", "o_proj",
                    "gate_proj", "up_proj", "down_proj"],
    lora_dropout=0.05, bias="none",
    task_type="CAUSAL_LM"
)
model = get_peft_model(model, lora_config)
model.print_trainable_parameters()

# STEP 4: Load Dataset
dataset = load_dataset("json", data_files="hindi_conversations.jsonl", split="train")

def format_prompt(example):
    return {"text": f"<bos><start>user<\n>{example['instruction']}<end>\n<start>model<\n>{example['response']}<end>"}

dataset = dataset.map(format_prompt)

# STEP 5: Train
training_args = TrainingArguments(
    output_dir="./gemma-hindi-bus",
    num_train_epochs=5,
    per_device_train_batch_size=4,
    gradient_accumulation_steps=4,
    learning_rate=2e-4,
    weight_decay=0.01,
    warmup_ratio=0.03,
    lr_scheduler_type="cosine",
    logging_steps=10,
    save_strategy="epoch",
    fp16=True,
    optim="paged_adamw_8bit",
)

trainer = SFTTrainer(
    model=model, train_dataset=dataset, tokenizer=tokenizer,
    args=training_args, dataset_text_field="text", max_seq_length=512,
)
trainer.train()

# STEP 6: Save LoRA + Merge
model.save_pretrained("./gemma-hindi-bus-lora")

from peft import PeftModel
base_model = AutoModelForCausalLM.from_pretrained("google/gemma-2b", torch_dtype=torch.float16, device_map="cpu")
merged = PeftModel.from_pretrained(base_model, "./gemma-hindi-bus-lora")
merged = merged.merge_and_unload()
merged.save_pretrained("./gemma-hindi-bus-merged")
tokenizer.save_pretrained("./gemma-hindi-bus-merged")

What is LLaVA?

Large Language-and-Vision Assistant — understands images + text together

  • Model: LLaVA 1.5 7B (open source)
  • Capability: Image understanding + text generation
  • Input: Screenshot + prompt like "What buttons are here?"
  • Output: Structured description of UI elements
How It Works for Us

1. Android app captures screen screenshot

2. Screenshot sent to LLaVA with structured prompt

3. LLaVA describes: buttons, text, list items, actions

4. Description converted to actionable commands

5. Actions sent back to Android app

def record_audio(duration=5, sample_rate=16000):
    """Record audio from microphone"""
    chunk = 1024
    format = pyaudio.paInt16
    channels = 1

    p = pyaudio.PyAudio()
    stream = p.open(format=format, channels=channels,
                    rate=sample_rate, input=True,
                    frames_per_buffer=chunk)

    print("Listening... Speak now!")
    frames = []
    for _ in range(0, int(sample_rate / chunk * duration)):
        data = stream.read(chunk)
        frames.append(data)

    stream.stop_stream()
    stream.close()
    p.terminate()

    # Save as WAV
    audio_path = "temp_recording.wav"
    with wave.open(audio_path, 'wb') as wf:
        wf.setnchannels(channels)
        wf.setsampwidth(p.get_sample_size(format))
        wf.setframerate(sample_rate)
        wf.writeframes(b''.join(frames))

    return audio_path

def speech_to_text(audio_path):

def read_screen(screenshot_path):
    """Read phone screen using LLaVA"""

    )
