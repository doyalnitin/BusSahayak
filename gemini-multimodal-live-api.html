1. WHAT IS GEMINI MULTIMODAL LIVE API
GOOGLE AI
Gemini Multimodal Live API
Google's real-time multimodal AI API that handles
audio, video, and text in a single WebSocket connection.

Key Features:
- Native audio understanding (no separate STT)
- Native audio generation (no separate TTS)
- Sub-500ms latency end-to-end
- Barge-in support (interrupt anytime)
- 29 languages including Hindi
- Function calling (book buses, search routes)
- Real-time streaming responses
WHY BETTER
vs Traditional STT+LLM+TTS
Traditional Pipeline:
Audio -> STT (Vosk) -> Text -> LLM (Gemma) -> Text -> TTS (Piper) -> Audio
Total: 3 separate models, 250ms+

Gemini Live API:
Audio -> Gemini (all-in-one) -> Audio
Total: 1 model, under 500ms, native understanding

Advantages:
- Understands tone, emotion, context
- Handles interruptions naturally
- No model switching overhead
- Function calling built-in
- Self-contained (no separate STT/TTS)

# main.py - FastAPI with Gemini Multimodal Live API import asyncio, json, os, base64 from fastapi import FastAPI, WebSocket, WebSocketDisconnect from fastapi.middleware.cors import CORSMiddleware import google.generativeai as genai app = FastAPI(title="BusSahayak Gemini Voice API") app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"]) # Configure Gemini genai.configure(api_key=os.getenv("GEMINI_API_KEY")) # System prompt for BusSahayak SYSTEM_PROMPT = """ You are BusSahayak, a voice assistant for blind and visually impaired users booking buses in India. You speak in Hindi (Hinglish is fine). Rules: 1. Be concise - 1-2 sentences max 2. Always confirm before booking 3. Use function calls for bus search, seat selection, booking 4. Read out bus options as numbered list 5. If user says a number, select that option 6. Be patient and repeat if user asks 7. Say "Samajh gaya" to confirm understanding 8. Use simple Hindi words Functions available: - search_buses(from_city, to_city, date) -> returns bus list - get_seats(bus_id) -> returns seat availability - create_booking(bus_id, seat_id, passenger_name, phone) -> confirms booking """ # Define function schemas for Gemini TOOLS = [{ "function_declarations": [ { "name": "search_buses", "description": "Search buses between two cities", "parameters": { "type": "OBJECT", "properties": { "from_city": {"type": "STRING", "description": "Source city"}, "to_city": {"type": "STRING", "description": "Destination city"}, "date": {"type": "STRING", "description": "Travel date"} }, "required": ["from_city", "to_city"] } }, { "name": "create_booking", "description": "Book a bus ticket", "parameters": { "type": "OBJECT", "properties": { "bus_id": {"type": "STRING"}, "seat_id": {"type": "STRING"}, "passenger_name": {"type": "STRING"}, "phone": {"type": "STRING"} }, "required": ["bus_id", "seat_id", "passenger_name", "phone"] } } ] }] # Execute function calls async def execute_function(name, args): if name == "search_buses": # Call ZuelPay API or mock data buses = search_buses_api(args["from_city"], args["to_city"]) return buses elif name == "create_booking": # Call ZuelPay booking API result = create_booking_api(args) return result # WebSocket endpoint for real-time voice @app.websocket("/ws/voice") async def voice_websocket(websocket: WebSocket): await websocket.accept() # Initialize Gemini Live session model = genai.GenerativeModel( model_name="gemini-2.0-flash-exp", system_instruction=SYSTEM_PROMPT, tools=TOOLS ) async with model.start_chat() as chat: try: while True: # Receive audio from React Native app data = await websocket.receive_text() message = json.loads(data) if message["type"] == "audio": # Decode base64 audio audio_bytes = base64.b64decode(message["data"]) # Send to Gemini response = await chat.send_message_async( genai.protos.Part(inline_data=genai.protos.Blob( mime_type="audio/wav", data=audio_bytes )) ) # Check for function calls if response.candidates[0].content.parts: for part in response.candidates[0].content.parts: if part.function_call: # Execute function result = await execute_function( part.function_call.name, dict(part.function_call.args) ) # Send result back to Gemini response = await chat.send_message_async( genai.protos.Part(function_response=genai.protos.FunctionResponse( name=part.function_call.name, response={"result": result} )) ) # Send audio response back to app if response.candidates[0].content.parts: for part in response.candidates[0].content.parts: if part.inline_data: await websocket.send_text(json.dumps({ "type": "audio", "data": base64.b64encode(part.inline_data.data).decode() })) if part.text: await websocket.send_text(json.dumps({ "type": "text", "data": part.text })) except WebSocketDisconnect: pass
3.2 React Native Frontend - WebSocket Client
// hooks/useGeminiVoice.ts - React Native WebSocket Client import { useState, useRef, useCallback } from 'react'; import { Audio } from 'expo-av'; export function useGeminiVoice() { const [isListening, setIsListening] = useState(false); const [response, setResponse] = useState(''); const [isSpeaking, setIsSpeaking] = useState(false); const wsRef = useRef(null); const recordingRef = useRef(null); // Connect to Gemini WebSocket const connect = useCallback(() => { wsRef.current = new WebSocket('wss://your-server.com/ws/voice'); wsRef.current.onmessage = async (event) => { const msg = JSON.parse(event.data); if (msg.type === 'audio') { // Play audio response const audio = new Audio.Sound(); await audio.loadAsync({ uri: `data:audio/wav;base64,${msg.data}` }); await audio.playAsync(); setIsSpeaking(false); } if (msg.type === 'text') { setResponse(msg.data); } }; }, []); // Start listening const startListening = useCallback(async () => { connect(); setIsListening(true); // Start recording const { recording } = await Audio.Recording.createAsync( Audio.RecordingOptionsPresets.HIGH_QUALITY ); recordingRef.current = recording; // Stream audio chunks every 100ms const interval = setInterval(async () => { if (recordingRef.current) { const uri = recordingRef.current.getURI(); const response = await fetch(uri); const blob = await response.blob(); const reader = new FileReader(); reader.onload = () => { const base64 = reader.result.split(',')[1]; wsRef.current?.send(JSON.stringify({ type: 'audio', data: base64 })); }; reader.readAsDataURL(blob); } }, 100); // Store interval for cleanup recordingRef.current._interval = interval; }, [connect]); // Stop listening const stopListening = useCallback(async () => { if (recordingRef.current) { clearInterval(recordingRef.current._interval); await recordingRef.current.stopAndUnloadAsync(); recordingRef.current = null; } setIsListening(false); setIsSpeaking(true); }, []); return { isListening, isSpeaking, response, startListening, stopListening }; }
