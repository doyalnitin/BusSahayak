Gemini with Function Calling (BusSahayak)
Define tools that Gemini can call
from google import genai
from google.genai import types

client = genai.Client(api_key="YOUR_API_KEY")

# Define bus search function
search_buses = types.FunctionDeclaration(
    name="search_buses",
    description="Search buses between two cities",
    parameters=types.Schema(
        type=types.Type.OBJECT,
        properties={
            "from_city": types.Schema(type=types.Type.STRING),
            "to_city": types.Schema(type=types.Type.STRING),
            "date": types.Schema(type=types.Type.STRING),
        },
        required=["from_city", "to_city"]
    )
)

# Add to model
config = types.GenerateContentConfig(
    tools=[types.Tool(function_declarations=[search_buses])]
)

# Chat
response = client.models.generate_content(
    model="gemini-2.0-flash",
    contents="Mujhe Delhi se Jaipur jana hai kal",
    config=config
)
print(response.function_calls)

Create Tables
Run this SQL to create seat_cache and bookings tables

psql -U postgres -d bus_sahayak

-- Seat cache table (ZuelPay data)
CREATE TABLE seat_cache (
    id SERIAL PRIMARY KEY,
    bus_id VARCHAR(50) NOT NULL,
    seat_number VARCHAR(10) NOT NULL,
    seat_type VARCHAR(20) DEFAULT 'seater',
    status VARCHAR(20) DEFAULT 'available',
    price DECIMAL(10,2),
    last_refreshed TIMESTAMP DEFAULT NOW()
);

-- Bookings table
CREATE TABLE bookings (
    id SERIAL PRIMARY KEY,
    user_id VARCHAR(50) NOT NULL,
    bus_id VARCHAR(50) NOT NULL,
    seat_numbers TEXT[],
    status VARCHAR(20) DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT NOW(),
    confirmed_at TIMESTAMP,
    expires_at TIMESTAMP
);

-- Index for fast lookup
CREATE INDEX idx_seat_bus ON seat_cache(bus_id);
CREATE INDEX idx_booking_user ON bookings(user_id);
CREATE INDEX idx_booking_status ON bookings(status);

4
Redis Seat Lock — Full Flow
Complete seat lock logic for BusSahayak
import redis
import time

r = redis.Redis(host='localhost', port=6379, decode_responses=True)

def lock_seat(bus_id, seat_number, user_id):
    """Atomic seat lock — 60 min TTL"""
    key = f"bus:{bus_id}:seat:{seat_number}"

    # SETNX: Set if Not Exists (atomic!)
    locked = r.set(key, user_id, nx=True, ex=3600)

    if locked:
        return {"status": "locked", "ttl": 3600}
    else:
        current_user = r.get(key)
        ttl = r.ttl(key)
        return {"status": "already_locked", "by": current_user, "ttl": ttl}

def confirm_booking(bus_id, seat_number, user_id):
    """User confirmed — lock until booking is final"""
    result = lock_seat(bus_id, seat_number, user_id)
    if result["status"] == "locked":
        # User said "Haan confirm" — seat is now theirs
        return {"message": "Seat confirmed! Manager will call you."}
    return {"message": "Seat already taken by someone else"}

# Usage
result = lock_seat("bus_123", "L1", "user_nitin")
print(result)  # {'status': 'locked', 'ttl': 3600}

result = lock_seat("bus_123", "L1", "user_rahul")
print(result)  # {'status': 'already_locked', 'by': 'user_nitin', 'ttl': 3599}
