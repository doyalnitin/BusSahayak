1.2 Audio Conversion Commands
# Convert M4A to WAV (16kHz, Mono, 16-bit) for STT ffmpeg -i input.m4a -ar 16000 -ac 1 -sample_fmt s16 output_stt.wav # Convert M4A to WAV (48kHz, Mono, 16-bit) for TTS ffmpeg -i input.m4a -ar 48000 -ac 1 -sample_fmt s16 output_tts.wav # Batch convert all M4A files in a folder for f in *.m4a; do ffmpeg -i "$f" -ar 16000 -ac 1 -sample_fmt s16 "${f%.m4a}_stt.wav" done # Verify audio properties ffprobe -v error -show_entries stream=sample_rate,channels,bits_per_sample,sample_fmt output.wav

3.4 FastAPI Application Code
# main.py — FastAPI Application from fastapi import FastAPI, UploadFile, File from fastapi.responses import JSONResponse, StreamingResponse import requests, json, time, io app = FastAPI(title="BusSahayak Azure API") # Azure credentials (from environment variables) SPEECH_KEY = os.environ["SPEECH_KEY"] SPEECH_REGION = os.environ["SPEECH_REGION"] COSMOS_ENDPOINT = os.environ["COSMOS_ENDPOINT"] COSMOS_KEY = os.environ["COSMOS_KEY"] # ===== STT: Azure Custom Speech ===== def speech_to_text(audio_data): url = f"https://{SPEECH_REGION}.stt.speech.microsoft.com/speech/recognition/custom/cognitiveservices/v1" headers = { "Ocp-Apim-Subscription-Key": SPEECH_KEY, "Content-Type": "audio/wav; codecs=audio/pcm; samplerate=16000" } params = {"language": "hi-IN", "format": "detailed"} response = requests.post(url, headers=headers, params=params, data=audio_data) result = response.json() return result["DisplayText"] # ===== LLM: Azure OpenAI GPT-4 ===== def process_with_llm(text, language): url = "https://your-openai.openai.azure.com/openai/deployments/gpt-4/chat/completions?api-version=2024-02-15-preview" headers = {"api-key": "your-openai-key", "Content-Type": "application/json"} system_prompt = "आप BusSahayak हैं। हिंदी में छोटे जवाब दें।" if language == "hi" else "You are BusSahayak. Reply in English briefly." data = {"messages": [{"role": "system", "content": system_prompt}, {"role": "user", "content": text}], "max_tokens": 100} response = requests.post(url, headers=headers, json=data) return response.json()["choices"][0]["message"]["content"] # ===== TTS: Azure Custom Voice ===== def text_to_speech(text, language): url = f"https://{SPEECH_REGION}.tts.speech.microsoft.com/cognitiveservices/v1" headers = {"Ocp-Apim-Subscription-Key": SPEECH_KEY, "Content-Type": "application/ssml+xml", "X-Microsoft-OutputFormat": "riff-24khz-16bit-mono-pcm"} voice = "hi-IN-NeuralFemale" if language == "hi" else "en-US-NeuralFemale" ssml = f"<speak version='1.0' xmlns='http://www.w3.org/2001/10/synthesis' xml:lang='{language}'><voice name='{voice}'>{text}</voice></speak>" response = requests.post(url, headers=headers, data=ssml) return response.content # ===== Main Endpoint ===== @app.post("/voice") async def process_voice(audio: UploadFile = File(...)): start = time.time() audio_data = await audio.read() user_text = speech_to_text(audio_data) language = "hi" if any(ord(c) > 0x0900 for c in user_text) else "en" response_text = process_with_llm(user_text, language) audio_response = text_to_speech(response_text, language) latency = int((time.time() - start) * 1000) return JSONResponse(content={"user_text": user_text, "response_text": response_text, "latency_ms": latency})

PHASE 5: INTEGRATE (Continuous)
5.1 Complete Request Flow
Step 1: User Speaks
Blind user holds screen → Microphone activates → Records 16kHz PCM audio → Sends to Azure App Service via HTTPS
Step 2: Audio → Azure Custom STT
FastAPI receives audio → Sends to centralindia.stt.speech.microsoft.com → Custom model transcribes → Returns Hindi/English text → Latency: 100-200ms
Step 3: Text → Azure OpenAI (GPT-4)
FastAPI sends text to your-openai.openai.azure.com → GPT-4 processes intent → Generates response in same language → Latency: 200-300ms
Step 4: Response → Azure Custom TTS
FastAPI sends response text to centralindia.tts.speech.microsoft.com → Custom voice generates 24kHz PCM audio → Latency: 100-200ms
Step 5: Audio → User
FastAPI returns audio to React Native app → Speaker plays response → Haptic vibration confirms → Total Latency: 400-600ms
Step 6: Data → Cosmos DB + Redis
Booking saved to Cosmos DB bookings container → Seat locked in Redis with 3600s TTL → Voice log saved to Cosmos DB voice_logs container → FCM push to manager app

5.3 Language Auto-Detection
def detect_language(text): # Count Hindi characters (Devanagari: 0900-097F) hindi_chars = len([c for c in text if 0x0900 <= ord(c) <= 0x097F]) # Count English characters english_chars = len([c for c in text if c.isalpha() and ord(c) < 128]) total = hindi_chars + english_chars if total == 0: return "hi" if hindi_chars / total > 0.7: return "hi" if english_chars / total > 0.7: return "en" return "hi" # Hinglish → treat as Hindi
5.4 React Native Integration
// hooks/useAzureVoice.ts import { useRef, useState } from 'react'; export function useAzureVoice() { const [isListening, setIsListening] = useState(false); const [response, setResponse] = useState(''); const mediaRecorder = useRef(null); const startListening = async () => { setIsListening(true); const stream = await navigator.mediaDevices.getUserMedia({ audio: true }); mediaRecorder.current = new MediaRecorder(stream); const chunks = []; mediaRecorder.current.ondataavailable = (e) => chunks.push(e.data); mediaRecorder.current.onstop = async () => { const audioBlob = new Blob(chunks, { type: 'audio/wav' }); const formData = new FormData(); formData.append('audio', audioBlob); const res = await fetch('https://BusSahayakAPI.azurewebsites.net/voice', { method: 'POST', body: formData }); const data = await res.json(); setResponse(data.response_text); setIsListening(false); }; mediaRecorder.current.start(); }; const stopListening = () => { mediaRecorder.current?.stop(); }; return { isListening, response, startListening, stopListening }; }
